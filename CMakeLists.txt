cmake_minimum_required(VERSION 3.28...4.4)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Recording Actions supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_FRONTEND_API "Use OBS frontend" ON)
option(ENABLE_QT "Use OBS Qt widgets" ON)
include(compilerconfig)
include(defaults)
include(helpers)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_library(recording-files STATIC src/file-actions.cpp)
target_include_directories(recording-files PUBLIC src)
target_compile_definitions(recording-files PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin-main.cpp src/recording-actions.cpp src/settings.cpp src/updates.cpp)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE recording-files OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network)
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})
install(FILES "$<TARGET_FILE:Qt6::QSchannelBackendPlugin>" DESTINATION "${_name}/data/qt/tls")
qt_add_resources(${CMAKE_PROJECT_NAME} locale PREFIX "/recording-actions" BASE data FILES data/locale/en-US.ini data/locale/de-DE.ini)
include(CTest)
if(BUILD_TESTING)
  add_executable(recording-actions-tests tests/core-tests.cpp)
  target_link_libraries(recording-actions-tests PRIVATE recording-files)
  add_test(NAME recording-actions-safety COMMAND recording-actions-tests)
  add_executable(recording-actions-integration tests/integration-tests.cpp src/recording-actions.cpp src/settings.cpp src/updates.cpp)
  target_link_libraries(recording-actions-integration PRIVATE recording-files OBS::libobs Qt6::Widgets Qt6::Network)
  target_include_directories(recording-actions-integration PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "$<TARGET_PROPERTY:OBS::obs-frontend-api,INTERFACE_INCLUDE_DIRECTORIES>")
  qt_add_resources(recording-actions-integration locale PREFIX "/recording-actions" BASE data FILES data/locale/en-US.ini data/locale/de-DE.ini)
  add_test(NAME recording-actions-integration COMMAND recording-actions-integration -platform windows)
  string(JSON deps_version GET ${buildspec} dependencies prebuilt version)
  set_tests_properties(recording-actions-integration PROPERTIES
    TIMEOUT 45
    ENVIRONMENT "QT_PLUGIN_PATH=$<TARGET_FILE_DIR:Qt6::Core>/../plugins"
    ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:$<TARGET_FILE_DIR:OBS::libobs>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>;PATH=path_list_prepend:${CMAKE_CURRENT_SOURCE_DIR}/.deps/obs-deps-${deps_version}-x64/bin")
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_PACKAGE_FILE_NAME "${_name}-${PROJECT_VERSION}-windows-x64")
include(CPack)
